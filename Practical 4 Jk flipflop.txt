Practical 4 Jk flipflop



module jk_flipflop (
    input J,
    input K,
    input clk,
    input RESET,
    output reg Q
);

always @(posedge clk or posedge RESET)
begin
    if (RESET)
        Q <= 1'b0;

    else
    begin
        case ({J,K})

            2'b00: Q <= Q;     // No Change
            2'b01: Q <= 1'b0;  // Reset
            2'b10: Q <= 1'b1;  // Set
            2'b11: Q <= ~Q;    // Toggle

        endcase
    end
end

endmodule

module jk_flipflop_tb;

reg J, K, clk, RESET;
wire Q;

jk_flipflop uut (
    .J(J),
    .K(K),
    .clk(clk),
    .RESET(RESET),
    .Q(Q)
);

// Clock generation
always #5 clk = ~clk;

initial
begin
    $monitor("Time=%0t | J=%b K=%b RESET=%b | Q=%b",
              $time, J, K, RESET, Q);

    // Initial values
    clk = 0;
    RESET = 1;
    J = 0;
    K = 0;

    #10 RESET = 0;

    // No Change
    J = 0; K = 0; #10;

    // Reset
    J = 0; K = 1; #10;

    // Set
    J = 1; K = 0; #10;

    // Toggle
    J = 1; K = 1; #10;

    // Toggle again
    J = 1; K = 1; #10;

    $finish;
end

endmodule
