Practical 2 Full adder using two half adder


module half_adder(
    input A,
    input B,
    output Sum,
    output Carry
);

assign Sum = A ^ B;
assign Carry = A & B;

endmodule

module full_adder(
    input A,
    input B,
    input Cin,
    output Sum,
    output Cout
);

wire S1, C1, C2;

// First Half Adder
half_adder HA1(
    .A(A),
    .B(B),
    .Sum(S1),
    .Carry(C1)
);

// Second Half Adder
half_adder HA2(
    .A(S1),
    .B(Cin),
    .Sum(Sum),
    .Carry(C2)
);

// OR gate
assign Cout = C1 | C2;

endmodule

module full_adder_tb;

reg A, B, Cin;
wire Sum, Cout;

// Instantiate Full Adder
full_adder uut(
    .A(A),
    .B(B),
    .Cin(Cin),
    .Sum(Sum),
    .Cout(Cout)
);

initial begin

    $monitor("A=%b B=%b Cin=%b | Sum=%b Cout=%b",
              A, B, Cin, Sum, Cout);

    A=0; B=0; Cin=0; #10;
    A=0; B=0; Cin=1; #10;
    A=0; B=1; Cin=0; #10;
    A=0; B=1; Cin=1; #10;
    A=1; B=0; Cin=0; #10;
    A=1; B=0; Cin=1; #10;
    A=1; B=1; Cin=0; #10;
    A=1; B=1; Cin=1; #10;

    $finish;

end

endmodule
