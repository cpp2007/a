Practical 1 =4:1 mux



module mux4to1 (
    input I0,
    input I1,
    input I2,
    input I3,
    input S1,
    input S0,
    output reg Y
);

always @(*) begin
    case ({S1, S0})

        2'b00: Y = I0;
        2'b01: Y = I1;
        2'b10: Y = I2;
        2'b11: Y = I3;

        default: Y = 1'b0;

    endcase
end

endmodule

module mux4to1_tb;

reg I0, I1, I2, I3;
reg S1, S0;
wire Y;

// Instantiate DUT
mux4to1 uut (
    .I0(I0),
    .I1(I1),
    .I2(I2),
    .I3(I3),
    .S1(S1),
    .S0(S0),
    .Y(Y)
);

initial begin

    $monitor("S1=%b S0=%b | I0=%b I1=%b I2=%b I3=%b | Y=%b",
              S1, S0, I0, I1, I2, I3, Y);

    I0=1; I1=0; I2=1; I3=0;

    S1=0; S0=0; #10;
    S1=0; S0=1; #10;
    S1=1; S0=0; #10;
    S1=1; S0=1; #10;

    $finish;

end

endmodule
